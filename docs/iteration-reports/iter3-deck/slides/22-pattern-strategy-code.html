Strategy는 “환불 계산식”을 정책 클래스로 분리한다

RefundHandler는 어떤 정책인지 고른 뒤, 계산은 RefundPolicy 인터페이스에 맡긴다. 그래서 발표에서는 paid가 결제 총액이고, concrete policy마다 반환식이 달라진다고 설명하면 된다.

// RefundHandler.java — Context
RefundPolicy policy = resolvePolicy(fareRule);
BigDecimal paid = BigDecimal.ZERO;
for (Payment p : reservation.getPayments()) {
    if (p != null && p.getAmount() != null) {
        paid = paid.add(p.getAmount());
    }
}
BigDecimal refundAmount =
    policy.calculateRefundAmount(paid);

private RefundPolicy resolvePolicy(FareRule rule) {
    if (rule == null || !rule.isRefundable()) {
        return new NoRefundPolicy();
    }
    if ("Y".equals(rule.getFareClass())
        || "B".equals(rule.getFareClass())) {
        return new FullRefundPolicy();
    }
    return new PartialRefundPolicy();
}

// RefundPolicy.java — Strategy
interface RefundPolicy {
    BigDecimal calculateRefundAmount(BigDecimal paid);
}
