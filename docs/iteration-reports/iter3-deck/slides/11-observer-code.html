public Payment confirmPayment(Reservation r,
                              FareRule rule,
                              long baseFare,
                              long tax) {
    if (!paymentProcessor.validateFareRule(rule)) {
        throw new IllegalArgumentException(...);
    }
    long total = paymentProcessor.calculateTotalAmount(baseFare, tax);
    Payment payment = paymentProcessor.processPaymentCharge(total);

    if (payment.getStatus() == PaymentStatus.PAID) {
        r.addPayment(payment);
        r.processPayment();
    } else {
        // 호출자가 직접 전이를 책임진다
        r.handlePaymentFailure();
    }
    return payment;
}
문제 — 동일한 "실패 시 Cancelled 전이"가 마일리지 결제, 좌석 hold 만료, 항공편 취소에서도 필요. 매번 호출자에서 같은 if/else를 반복해야 한다.

public class PaymentProcessor extends EventPublisher {
  public Payment processPaymentCharge(long amount, String pnr) {

    if (authorized) {
        payment.pay();

        payment.fail();
        publish(new PaymentFailedEvent(
            payment, pnr, "gateway-declined"));
    }
    return payment;
  }

// ReservationAutoCancelListener — Observer
public class ReservationAutoCancelListener
       implements EventListener {
  public void onEvent(DomainEvent e) {
    if (!(e instanceof PaymentFailedEvent ev)) return;
    Reservation r = Reservation.findByPnr(ev.getReservationPnr());
    if (r != null) r.handlePaymentFailure();
  }
}
변화 — BookingController에서 handlePaymentFailure() 직접 호출 사라짐. 새 발생원(마일리지 실패, 항공권 구매 후 버스티켓 발매)도 publisher가 이벤트만 발행하면 listener가 동일 구조로 반응.
