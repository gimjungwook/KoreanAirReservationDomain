Subject는 listener 구현체를 모른다
// EventPublisher.java — Subject base
private final List<EventListener> listeners =
    new ArrayList<>();

public void subscribe(EventListener listener) {
    if (listener == null || listeners.contains(listener)) return;
    listeners.add(listener);
}

protected void publish(DomainEvent event) {
    for (EventListener listener : new ArrayList<>(listeners)) {
        listener.onEvent(event);
    }
}

// TicketPurchasePublisher.java — ConcreteSubject
public void publishTicketIssued(Reservation r,
                              Ticket ticket,
                              BusCity city) {
    publish(new TicketIssuedEvent(r, ticket, city));
}

// BusTicketPurchaseListener.java — ConcreteObserver
public void onEvent(DomainEvent event) {
    if (!(event instanceof TicketIssuedEvent e)) return;
    if (e.getRequestedBusCity() == null) return;
    busTicketingService.issuePremiumTicket(
        e.getReservation(),
        e.getTicket(),
        e.getRequestedBusCity());
}
