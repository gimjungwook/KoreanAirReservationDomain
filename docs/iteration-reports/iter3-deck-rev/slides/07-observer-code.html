domain/event/EventPublisher.javaSUBJECT
public class EventPublisher {
  private final List<EventListener> listeners
      = new ArrayList<>();

  public void subscribe(EventListener l) {
    if (l == null || listeners.contains(l)) return;
    listeners.add(l);
  }

  protected void publish(DomainEvent event) {
    for (EventListener l : new ArrayList<>(listeners)) {
      try { l.onEvent(event); }            // 1-to-N broadcast
      catch (RuntimeException ex) {        // 한 listener 실패가
        log(l, event, ex);                // 다른 통지를 끊지 않음
      }
    }
  }
}

public class ReservationAutoCancelListener
    implements EventListener {

  @Override
  public void onEvent(DomainEvent event) {
    if (!(event instanceof PaymentFailedEvent))
      return;                            // 자기 타입만 처리
    PaymentFailedEvent e = (PaymentFailedEvent) event;
    Reservation r = Reservation.findByPnr(e.getReservationPnr());
    if (r == null) return;
    r.handlePaymentFailure();             // iter1 State 전이 호출
    println("[AUTO-CANCEL] " + e.getReason());
  }
}
