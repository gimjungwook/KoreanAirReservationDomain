// UC "Search Connecting Flights" 의 실현 — include Book Flight
public List<Itinerary> searchConnecting(String from, String to,
                                       LocalDate date, Duration mct) {
  List<Itinerary> result = new ArrayList<>();
  Duration min = mct != null ? mct : Itinerary.INTERNATIONAL_MCT;
  for (FlightSchedule a : catalog) {           // 1구간 후보
    if (!isFirstLegOf(a, from, date)) continue;
    String hub = destinationCode(a);
    for (FlightSchedule b : catalog) {         // 2구간 후보
      if (!hub.equalsIgnoreCase(originCode(b))) continue;
      if (!to.equalsIgnoreCase(destinationCode(b))) continue;
      Itinerary c = Itinerary.connecting(a, b);   // 환승 일정 생성
      if (c.isConnectionTimeValid(min)) result.add(c); // MCT 통과만
    }
  }
  return result;
}
