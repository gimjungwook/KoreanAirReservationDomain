// 디폴트 = 모든 전이 거부. 구체 상태는 허용만 override.
public abstract class AbstractReservationState
    implements ReservationState {

  @Override public void processPayment(Reservation ctx) {
    throw new InvalidStateTransitionException(name(), "processPayment");
  }
  @Override public void handlePaymentFailure(Reservation ctx) {
    throw new InvalidStateTransitionException(name(), "handlePaymentFailure");
  }
  // requestCancellation, issueTicket, requestRefund ... 모두 디폴트 거부
}
state/PendingPaymentState.javaSTATE 구체
public class PendingPaymentState extends AbstractReservationState {

  @Override public String name() { return "PendingPayment"; }

  @Override public void processPayment(Reservation ctx) {
    ctx.setState(new ConfirmedState());        // 성공 → Confirmed
    ctx.updateStatus(ReservationStatus.CONFIRMED);
  }
  @Override public void handlePaymentFailure(Reservation ctx) {
    ctx.setState(new CancelledState());        // 실패 → Cancelled
    ctx.updateStatus(ReservationStatus.CANCELLED);
  }
}
Observer listener가 호출하는 handlePaymentFailure = 이 State 전이디폴트 거부 + 허용만 override = 상태별 가드 중복 제거
