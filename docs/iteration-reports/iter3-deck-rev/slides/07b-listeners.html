ReservationHoldListenerLISTENER
public void onEvent(DomainEvent event) {
  if (!(event instanceof SeatHoldExpiredEvent))
    return;
  SeatHoldExpiredEvent e =
      (SeatHoldExpiredEvent) event;
  Seat seat = e.getSeat();
  if (seat != null) {
    seat.release();            // 좌석 해제
    println("[HOLD-EXPIRY] released");
  }
  String pnr = e.getReservationPnr();
  if (pnr == null) return;
  Reservation r = Reservation.findByPnr(pnr);
  if (r == null) return;
  try {
    r.handlePaymentFailure();  // 예약 취소
  } catch (InvalidStateTransitionException ig) {
    println("전이 무시 " + r.getStateName());
  }
}
ReservationAutoCancelListenerLISTENER
public void onEvent(DomainEvent event) {
  if (!(event instanceof PaymentFailedEvent))
    return;
  PaymentFailedEvent e =
      (PaymentFailedEvent) event;
  String pnr = e.getReservationPnr();
  if (pnr == null) return;
  Reservation r = Reservation.findByPnr(pnr);
  if (r == null) return;
  try {
    r.handlePaymentFailure();  // PendingPayment
    println("[AUTO-CANCEL] "     //  → Cancelled
            + e.getReason());
  } catch (InvalidStateTransitionException ex) {
    println("전이 실패 " + r.getStateName());
  }
}
AffectedReservationListenerLISTENER
public void onEvent(DomainEvent event) {
  if (!(event instanceof FlightStatusChangedEvent))
    return;
  FlightStatusChangedEvent e =
      (FlightStatusChangedEvent) event;
  FlightSchedule sch = e.getSchedule();
  if (sch == null) return;
  List<Reservation> affected = findAffected(sch);
  for (Reservation r : affected) {  // 1-to-N 전파
    r.evaluateImpactOfFlightStatusChange();
    String tag = e.getNewStatus()
        == FlightStatus.CANCELLED
        ? "CANCEL" : "STATUS";
    println("[FLIGHT-" + tag + "] PNR="
            + r.getPnrNumber());
  }
}
공통: instanceof 가드 → payload 추출 → 도메인 호출3종 모두 EventListener.onEvent 한 메서드 구현
